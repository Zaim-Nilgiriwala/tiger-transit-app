```typescript
  longitude: number;

export interface VehiclePosition {
  vehicleId: string;
  routeId: string;
  lat: number;
  lon: number;
  heading: number;
  speed: number;
  load: number;
  capacity: number;
  nextStopId: string;
  etaSeconds: number;
  onTime: number;
  lastStopId: string;
  isDelayed: boolean;
  timestamp: number;
}

export interface ArrivalPrediction {
  vehicleId: string;
  routeId: string;
  stopId: string;
  etaSeconds: number;
  isModelPrediction: boolean;
}

export interface ServiceAlert {
  id: string;
  headerText: string;
  descriptionText: string;
  activePeriodStart: number;
  activePeriodEnd: number;
  routeIds: string[];
}
```

Create 6 Redux slices using @reduxjs/toolkit createSlice:

**routesSlice.ts**: Initial state: `{ list: [], stops: {}, shapes: {}, loading: false, error: null }`. Types: list is Route[], stops is Record<string, Stop[]>, shapes is Record<string, Coordinate[]>. Add reducers: setRoutes, setRouteStops, setRouteShape, setLoading, setError.

**vehiclesSlice.ts**: Initial state: `{ positions: [], lastUpdated: 0, connected: false }`. Types per VehiclePosition[]. Add reducers: setPositions, setConnected, clearPositions.

**predictionsSlice.ts**: Initial state: `{ byStop: {}, loading: false }`. Types: byStop is Record<string, ArrivalPrediction[]>. Add reducers: setPredictions, setLoading, clearPredictions.

**uiSlice.ts**: Initial state: `{ selectedRouteId: null, selectedStopId: null, sheetPosition: 'collapsed', showFavoritesOnly: false, activeCallout: null }`. Add reducers: selectRoute, selectStop, setSheetPosition, toggleFavoritesOnly, setActiveCallout, clearCallout.

**preferencesSlice.ts**: Initial state: `{ favoriteRouteIds: [], favoriteStopIds: [] }`. Add reducers: toggleFavoriteRoute, toggleFavoriteStop, setFavorites (for hydrating from AsyncStorage).

**alertsSlice.ts**: Initial state: `{ active: [], lastFetched: 0 }`. Types: active is ServiceAlert[]. Add reducers: setAlerts, clearAlerts.
